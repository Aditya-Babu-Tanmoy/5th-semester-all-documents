@Composable
fun WeatherPage(viewModel: WeatherViewModel) {

    var city by remember {
        mutableStateOf("")
    }

    val weatherResult by viewModel.weatherResult.observeAsState()

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(16.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {

        // =========================
        // SEARCH BAR
        // =========================

        Row(
            modifier = Modifier.fillMaxWidth(),
            verticalAlignment = Alignment.CenterVertically
        ) {

            OutlinedTextField(
                value = city,
                onValueChange = {
                    city = it
                },
                label = {
                    Text("Search")
                },
                singleLine = true,
                modifier = Modifier.weight(1f)
            )

            IconButton(
                onClick = {
                    if (city.isNotBlank()) {
                        viewModel.getData(city)
                    }
                }
            ) {
                Icon(
                    imageVector = Icons.Default.Search,
                    contentDescription = "Search",
                    modifier = Modifier.size(30.dp)
                )
            }
        }

        Spacer(modifier = Modifier.height(25.dp))


        // =========================
        // WEATHER RESULT
        // =========================

        when (val result = weatherResult) {

            // -------------------------
            // LOADING
            // -------------------------

            is NetworkResponse.Loading -> {

                Text(
                    text = "Loading...",
                    fontSize = 18.sp
                )
            }


            // -------------------------
            // SUCCESS
            // -------------------------

            is NetworkResponse.Success -> {

                val weather = result.data


                // =========================
                // LOCATION
                // =========================

                Row(
                    modifier = Modifier.fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically
                ) {

                    Text(
                        text = "📍",
                        fontSize = 28.sp
                    )

                    Spacer(modifier = Modifier.width(12.dp))

                    Text(
                        text = "${weather.location.name}, ${weather.location.country}",
                        fontSize = 20.sp,
                        fontWeight = FontWeight.Medium
                    )
                }


                Spacer(modifier = Modifier.height(25.dp))


                // =========================
                // TEMPERATURE
                // =========================

                Text(
                    text = "${weather.current.temp_c}°C",
                    fontSize = 55.sp,
                    fontWeight = FontWeight.Light
                )


                Spacer(modifier = Modifier.height(10.dp))


                // =========================
                // WEATHER ICON
                // =========================

                val iconUrl =
                    if (weather.current.condition.icon.startsWith("//")) {
                        "https:${weather.current.condition.icon}"
                    } else {
                        weather.current.condition.icon
                    }

                AsyncImage(
                    model = iconUrl,
                    contentDescription = weather.current.condition.text,
                    modifier = Modifier.size(100.dp)
                )


                // =========================
                // WEATHER CONDITION
                // =========================

                Text(
                    text = weather.current.condition.text,
                    fontSize = 20.sp,
                    fontWeight = FontWeight.Medium
                )


                Spacer(modifier = Modifier.height(25.dp))


                // =========================
                // WEATHER INFORMATION CARD
                // =========================

                Card(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(horizontal = 5.dp),
                    shape = RoundedCornerShape(16.dp),
                    border = BorderStroke(
                        1.dp,
                        Color.Gray
                    ),
                    colors = CardDefaults.cardColors(
                        containerColor = Color.Transparent
                    )
                ) {

                    Column(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(15.dp)
                    ) {


                        // =========================
                        // FIRST ROW
                        // =========================

                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.SpaceEvenly
                        ) {

                            // HUMIDITY

                            WeatherInfoItem(
                                icon = {
                                    Icon(
                                        imageVector = Icons.Default.WaterDrop,
                                        contentDescription = "Humidity",
                                        tint = Color(0xFF64B5F6),
                                        modifier = Modifier.size(30.dp)
                                    )
                                },
                                title = "Humidity",
                                value = "${weather.current.humidity}%"
                            )


                            // WIND

                            WeatherInfoItem(
                                icon = {
                                    Icon(
                                        imageVector = Icons.Default.Air,
                                        contentDescription = "Wind",
                                        tint = Color(0xFF64B5F6),
                                        modifier = Modifier.size(30.dp)
                                    )
                                },
                                title = "Wind",
                                value = "${weather.current.wind_kph} km/h"
                            )
                        }


                        Spacer(modifier = Modifier.height(20.dp))


                        // =========================
                        // SECOND ROW
                        // =========================

                        Row(
                            modifier = Modifier.fillMaxWidth(),
                            horizontalArrangement = Arrangement.SpaceEvenly
                        ) {

                            // UV

                            WeatherInfoItem(
                                icon = {
                                    Icon(
                                        imageVector = Icons.Default.WbSunny,
                                        contentDescription = "UV",
                                        tint = Color(0xFFFFB74D),
                                        modifier = Modifier.size(30.dp)
                                    )
                                },
                                title = "UV Index",
                                value = weather.current.uv
                            )


                            // PRECIPITATION

                            WeatherInfoItem(
                                icon = {
                                    Icon(
                                        imageVector = Icons.Default.Umbrella,
                                        contentDescription = "Precipitation",
                                        tint = Color(0xFF9575CD),
                                        modifier = Modifier.size(30.dp)
                                    )
                                },
                                title = "Precipitation",
                                value = "${weather.current.precip_in} in"
                            )
                        }
                    }
                }
            }


            // -------------------------
            // ERROR
            // -------------------------

            is NetworkResponse.Error -> {

                Text(
                    text = result.message,
                    fontSize = 18.sp
                )
            }


            // -------------------------
            // INITIAL STATE
            // -------------------------

            null -> {

                Text(
                    text = "Search for a city",
                    fontSize = 18.sp
                )
            }
        }
    }
}


// ==================================================
// WEATHER INFORMATION ITEM
// ==================================================

@Composable
fun WeatherInfoItem(
    icon: @Composable () -> Unit,
    title: String,
    value: String
) {

    Row(
        modifier = Modifier
            .width(140.dp),
        verticalAlignment = Alignment.CenterVertically
    ) {

        icon()

        Spacer(modifier = Modifier.width(10.dp))

        Column {

            Text(
                text = title,
                fontSize = 14.sp
            )

            Text(
                text = value,
                fontSize = 16.sp,
                fontWeight = FontWeight.Bold
            )
        }
    }
}
